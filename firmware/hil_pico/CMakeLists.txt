cmake_minimum_required(VERSION 3.13)

if(DEFINED PICO_SDK_PATH)
    include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
elseif(DEFINED ENV{PICO_SDK_PATH})
    include("$ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake")
else()
    message(FATAL_ERROR "Set PICO_SDK_PATH to the Raspberry Pi Pico SDK checkout")
endif()

project(hil_plant_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

add_executable(hil_plant_pico src/main.c)
target_include_directories(hil_plant_pico PRIVATE ../common)
target_link_libraries(hil_plant_pico pico_stdlib pico_unique_id hardware_uart)
pico_enable_stdio_usb(hil_plant_pico 1)
pico_enable_stdio_uart(hil_plant_pico 0)
pico_add_extra_outputs(hil_plant_pico)
